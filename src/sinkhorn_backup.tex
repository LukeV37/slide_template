  \vspace{14mm}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.45\textwidth}
      \begin{lstlisting}[style=pythoncode]
if epoch < warmup:
    alignment_scale = 0
elif epoch < warmup + ramp:
    alignment_scale increases gradually
else:
    alignment_scale = 1
      \end{lstlisting}
    \end{column}
    \begin{column}{0.51\textwidth}
      \begin{itemize}\tightlist
        \item Warmup keeps optimization classification-only.
        \item Ramp avoids abruptly applying the full OT penalty.
        \item After ramp, configured Sinkhorn \texttt{weight} is fully active.
      \end{itemize}
      \vspace{2mm}
      \begin{center}
        {\small\textbf{Conceptual objective}}\par
        \vspace{1mm}
        {\small $L = L_{\mathrm{CE, nominal}} + \mathrm{weight}\times\mathrm{scale}(t)\times D_{\mathrm{Sinkhorn}}$}
      \end{center}
    \end{column}
  \end{columns}
