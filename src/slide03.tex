  \begin{itemize}
    \item \textbf{Wafer cut}
    \begin{itemize}
      \item Restrict to wafer w206, manually chosen as a ``well-behaved'' wafer
    \end{itemize}
    \item \textbf{Unstable detector cut}
    \begin{itemize}
      \item Compute each detector's median, 10th percentile, and 90th percentile over time
      \item Keep only detectors with finite, positive median
      \item Require 10th percentile $> 0.9 \times$ median
      \item Require 90th percentile $< 1.1 \times$ median
      \item In short: keep detectors within about $\pm 10\%$ of median most of the time
    \end{itemize}
    \item \textbf{Bad timestamp cut}
    \begin{itemize}
      \item For each kept detector, define valid values as finite and positive
      \item Compute each detector's 1st percentile from valid samples
      \item Keep timestamp only if every kept detector is finite, positive, and above its 1st percentile
      \item In short: drop timestamps where any kept detector is invalid or unusually low
    \end{itemize}
  \end{itemize}}
